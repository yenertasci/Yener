import cv2 import requests import time from flask import Flask, Response from flask_cors import CORS app = Flask(__name__) CORS(app) # Frontend ile sorunsuz haberleşme için # Bilgisayar kamerasını başlat (0: Varsayılan Kamera) camera = cv2.VideoCapture(0) # OpenCV Yüz Tanıma Sınıflandırıcısı face_cascade = cv2.CascadeClassifier(cv2.data.haarcascades + 'haarcascade_frontalface_default.xml') # Telegram Bot Bilgileri (Kendi bot token ve chat id'ni buraya yazabilirsin) TELEGRAM_BOT_TOKEN = "YOUR_BOT_TOKEN" TELEGRAM_CHAT_ID = "YOUR_CHAT_ID" last_alert_time = 0 def send_telegram_alert(frame_bytes): if TELEGRAM_BOT_TOKEN == "YOUR_BOT_TOKEN": return url = f"https://api.telegram.org/bot{TELEGRAM_BOT_TOKEN}/sendPhoto" try: files = {'photo': ('alert.jpg', frame_bytes, 'image/jpeg')} data = {'chat_id': TELEGRAM_CHAT_ID, 'caption': '🚨 UYARI: Odada Hareket / Tanınmayan Kişi Algılandı!'} requests.post(url, files=files, data=data, timeout=3) except Exception as e: print("Telegram Bildirim Hatası:", e) def generate_frames(): global last_alert_time while True: success, frame = camera.read() if not success: break gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY) faces = face_cascade.detectMultiScale(gray, scaleFactor=1.1, minNeighbors=5, minSize=(30, 30)) for (x, y, w, h) in faces: cv2.rectangle(frame, (x, y), (x+w, y+h), (0, 255, 170), 2) cv2.putText(frame, "KISI ALGILANDI", (x, y-10), cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 170), 2) if len(faces) > 0 and (time.time() - last_alert_time > 10): ret, buffer = cv2.imencode('.jpg', frame) if ret: send_telegram_alert(buffer.tobytes()) last_alert_time = time.time() ret, buffer = cv2.imencode('.jpg', frame, [cv2.IMWRITE_JPEG_QUALITY, 70]) frame = buffer.tobytes() yield (b'--frame\r\nContent-Type: image/jpeg\r\n\r\n' + frame + b'\r\n') @app.route('/video_feed') def video_feed(): return Response(generate_frames(), mimetype='multipart/x-mixed-replace; boundary=frame') if __name__ == '__main__': print("Python Akış Sunucusu Başlatıldı: http://localhost:5000/video_feed") app.run(host='0.0.0.0', port=5000, threaded=True)
